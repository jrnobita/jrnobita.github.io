# Smart Navy AI — Germany
#
# WHAT VANILLA ALREADY DOES, AND WHY IT IS NOT ENOUGH
#
# Vanilla 1.14.1 GER.txt ships a plan called axis_stay_away_from_the_royal_navy
# which avoids region 18 (English Channel, value 600), 16 and 43 (both commented
# "around UK", value 100). So the idea is present. It fails in practice because
# of its enable block:
#
#     date < 1942.1.1
#     OR = {
#         GER = { naval_strength_ratio = { tag = ENG ratio < 0.5 } }
#         ... thirteen divisions_in_state checks on British coastal states ...
#     }
#     abort_when_not_enabled = yes
#
# Two switches turn it off exactly when it is still needed:
#
#   1. The hard date cap. On 1 January 1942 the plan aborts and never returns.
#      Every convoy Germany runs after that date is routed with no penalty on
#      the Channel or the Western Approaches, which is where the Royal Navy and
#      its submarines live for the whole rest of the game.
#   2. The strength ratio. It only holds while the Kriegsmarine is under half
#      the Royal Navy. A Germany that builds up — an AI on a high difficulty
#      bonus, or Germany under Plan Z — crosses 0.5, the plan aborts, and the
#      fleet walks into the Home Fleet.
#
# It is also incomplete: 16 and 43 are only weighted 100, which is a preference,
# not a refusal, next to the 600 used for the Channel.
#
# This file replaces the intent with something that does not expire. Values
# stack with vanilla's, so where both are active the avoidance is simply firmer.


# ---------------------------------------------------------------------------
# The Channel and the approaches to Britain are a killing ground.
#
# No date. The condition is the naval balance and nothing else, so the plan
# holds in 1944 exactly as it does in 1939 — and it releases if Germany really
# does build a fleet that can win, which vanilla's 0.5 cut-off never allows for.
#
# 1.5 rather than 1.0 because the defender in home waters has land-based air,
# short repair runs and better spotting. Parity at sea is a loss in the Channel.
# ---------------------------------------------------------------------------
snai_ger_the_channel_is_a_killing_ground = {
	allowed = {
		original_tag = GER
	}
	enable = {
		has_war_with = ENG
		enemies_naval_strength_ratio < 1.5
	}
	abort_when_not_enabled = yes

	# 18 — English Channel. Vanilla's own comment on value 600 is "avoid doing
	# missions as well", i.e. this is the level that stops the AI basing
	# missions there at all, not merely routing around it.
	ai_strategy = {
		type = naval_avoid_region
		id = 18
		value = 600
	}

	# 43 — Western Approaches. Where the Royal Navy concentrates its escorts and
	# where inbound British convoys converge. Vanilla weights this 100; a
	# preference is not enough when it is also the shortest path west.
	ai_strategy = {
		type = naval_avoid_region
		id = 43
		value = 500
	}

	# 16 — North Sea. Within reach of British land-based air for its whole width.
	ai_strategy = {
		type = naval_avoid_region
		id = 16
		value = 300
	}

	# 44 — Denmark Strait. The Atlantic breakout, and the reason Bismarck's
	# sortie was a one-way trip. Worth using, never worth loitering in.
	ai_strategy = {
		type = naval_avoid_region
		id = 44
		value = 150
	}
}


# ---------------------------------------------------------------------------
# The Baltic is the war economy.
#
# Swedish iron ore moves by sea. It is the one German convoy route that matters
# and the one stretch of water Germany actually controls, because the Danish
# Belts (207) bottle the Royal Navy out of it — vanilla ENG.txt avoids 9, 206
# and 207 at value 600 for exactly that reason.
#
# So: base the fleet there and escort there. Note there is deliberately no
# naval_avoid_region for 9, 206, 207 or 279 anywhere in this mod — those are the
# waters Germany is supposed to be in.
# ---------------------------------------------------------------------------
snai_ger_the_baltic_is_the_lifeline = {
	allowed = {
		original_tag = GER
	}
	enable = {
		has_war = yes
	}
	abort_when_not_enabled = yes

	# 9 — Baltic Sea.
	ai_strategy = {
		type = strike_force_home_base
		id = 9
		value = 120
	}

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_ESCORT"
		value = -70
	}

	ai_strategy = {
		type = role_ratio
		id = naval_escort
		value = 30
	}
}


# ---------------------------------------------------------------------------
# Raid the gaps, not the escorts.
#
# The regions below are the mid-ocean stretches: far from British air cover,
# far from the escort concentration in the Western Approaches, and squarely on
# the convoy tracks. This is where a submarine campaign is cheap, and it is the
# opposite end of the Atlantic from where vanilla lets the fleet wander.
#
# Raiding is also the correct mission for the weaker navy on the merits: a
# submarine that meets an escort group breaks off, so being outnumbered costs
# far less here than it does in a fleet action.
# ---------------------------------------------------------------------------
snai_ger_wolfpacks_in_the_gaps = {
	allowed = {
		original_tag = GER
	}
	enable = {
		has_war_with = ENG
	}
	abort_when_not_enabled = yes

	# 51 — Mid Atlantic Gap: the air gap, and historically where the wolfpacks
	# were safe from land-based patrol aircraft.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 51
		value = 200
	}

	# 58 — Central Atlantic Gap.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 58
		value = 180
	}

	# 56 — Mid Atlantic.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 56
		value = 150
	}

	# 61 — Cape Verde Plain. Vanilla ENG raids here itself, which is a fair
	# signal that the traffic is worth attacking.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 61
		value = 120
	}

	# 45 — Norwegian Sea: the Arctic convoy route to Murmansk.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 45
		value = 120
	}

	ai_strategy = {
		type = role_ratio
		id = naval_submarine
		value = 50
	}

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_RAIDING"
		value = -80
	}
}


# ---------------------------------------------------------------------------
# The Atlantic ports change the arithmetic.
#
# With the French Atlantic coast in hand, the Bay of Biscay stops being a
# transit risk and becomes the front door — a boat sails from Brest or Lorient
# instead of going the long way round Scotland. Note that Biscay (42) is never
# given a naval_avoid_region by this mod for exactly that reason.
# ---------------------------------------------------------------------------
snai_ger_atlantic_ports = {
	allowed = {
		original_tag = GER
	}
	enable = {
		has_war_with = ENG
		FRA = {
			has_capitulated = yes
		}
	}
	abort_when_not_enabled = yes

	# 47 — Iberian Coast, on the route south out of Biscay.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 47
		value = 140
	}

	# 62 — Gulf of Guinea: the Cape route traffic, a long way from any escort.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 62
		value = 100
	}

	ai_strategy = {
		type = role_ratio
		id = naval_submarine
		value = 25
	}
}


# ---------------------------------------------------------------------------
# Norway, once it is Germany's.
#
# The Narvik ore route runs down the Norwegian coast, and holding Norway also
# puts the fleet a long way north of the Home Fleet's usual station. Gated on
# Norway no longer existing rather than on a state id, because state ids are the
# one class of identifier this mod could not verify from the game files.
# ---------------------------------------------------------------------------
snai_ger_norwegian_ore_route = {
	allowed = {
		original_tag = GER
	}
	enable = {
		has_war_with = ENG
		NOT = {
			country_exists = NOR
		}
	}
	abort_when_not_enabled = yes

	# 174 — Norwegian Coast.
	ai_strategy = {
		type = strike_force_home_base
		id = 174
		value = 80
	}
}
