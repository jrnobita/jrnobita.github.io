# Smart Navy AI — Britain, the United States, Italy, the Soviet Union, France
#
# The two headline cases are Germany and Japan, but a doctrine that only knows
# about two countries is not a doctrine. Each block below is the same idea
# applied to a different geography: base the fleet where its own shipping is,
# refuse the water where the other side has land-based air, and raid where the
# escorts are not.


# ===========================================================================
# BRITAIN
# ===========================================================================
#
# Britain's navy is rarely the problem — Britain's convoys are. The Atlantic
# lifeline is long, and a Britain that loses it loses the war without ever
# losing a fleet action. Vanilla already bases the Home Fleet in the North Sea
# (strike_force_home_base id = 16) and already avoids the Baltic at value 600,
# both of which are right, so neither is repeated here.

snai_eng_the_lifeline_comes_first = {
	allowed = {
		original_tag = ENG
	}
	enable = {
		has_war = yes
		any_enemy_country = {
			has_navy_size = {
				type = submarine
				size > 20
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_ESCORT"
		value = -80
	}

	ai_strategy = {
		type = role_ratio
		id = naval_escort
		value = 55
	}

	ai_strategy = {
		type = unit_ratio
		id = convoy
		value = 25
	}

	# 43 — Western Approaches. Britain's escorts belong here, on top of the
	# convergence point of every inbound convoy. This is the same water the
	# German plan in this mod treats as a killing ground — which is the point:
	# it is dangerous for Germany precisely because it is Britain's.
	ai_strategy = {
		type = strike_force_home_base
		id = 43
		value = 100
	}
}


# ===========================================================================
# UNITED STATES
# ===========================================================================
#
# The American submarine campaign against Japanese shipping was the single most
# cost-effective naval effort of the war, and the AI almost never runs it. These
# are the regions the Japanese oil lane actually passes through — the mirror of
# the escort priorities in snai_11_japan.txt.

snai_usa_strangle_the_oil_lane = {
	allowed = {
		original_tag = USA
	}
	enable = {
		has_war_with = JAP
	}
	abort_when_not_enabled = yes

	# 75 — South China Sea.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 75
		value = 200
	}

	# 76 — East China Sea.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 76
		value = 180
	}

	# 78 — Philippine Sea.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 78
		value = 160
	}

	# 93 — Java Sea, at the southern end where the oil is loaded.
	ai_strategy = {
		type = naval_convoy_raid_region
		id = 93
		value = 140
	}

	ai_strategy = {
		type = role_ratio
		id = naval_submarine
		value = 45
	}

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_RAIDING"
		value = -70
	}
}

# The Sea of Japan and the Japanese coast are inside Japanese land-based air
# cover. Worth entering once the carriers can cover it; not worth entering to
# get there first.
snai_usa_wait_for_the_carriers = {
	allowed = {
		original_tag = USA
	}
	enable = {
		has_war_with = JAP
		NOT = {
			has_navy_size = {
				type = carrier
				size > 5
			}
		}
	}
	abort_when_not_enabled = yes

	# 79 — Sea of Japan.
	ai_strategy = {
		type = naval_avoid_region
		id = 79
		value = 350
	}

	# 90 — Coast of Japan.
	ai_strategy = {
		type = naval_avoid_region
		id = 90
		value = 300
	}
}


# ===========================================================================
# ITALY
# ===========================================================================
#
# Italy has a real fleet and no way to replace it, and its convoy route to North
# Africa is short, fixed and entirely inside its own air cover. That combination
# argues for a fleet that escorts the Libya run and declines the general action
# — the historical "fleet in being" argument, and also the mathematically
# correct one when your losses are permanent and the other side's are not.

snai_ita_escort_the_libya_run = {
	allowed = {
		original_tag = ITA
	}
	enable = {
		has_war = yes
	}
	abort_when_not_enabled = yes

	# 29 — Central Mediterranean Sea: the Sicily narrows, the Libya convoy route
	# and Malta, all in one region, all under Italian land-based air.
	ai_strategy = {
		type = strike_force_home_base
		id = 29
		value = 150
	}

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_ESCORT"
		value = -70
	}

	ai_strategy = {
		type = role_ratio
		id = naval_escort
		value = 35
	}
}

snai_ita_do_not_fight_the_royal_navy_in_the_open = {
	allowed = {
		original_tag = ITA
	}
	enable = {
		has_war_with = ENG
		enemies_naval_strength_ratio < 1
	}
	abort_when_not_enabled = yes

	# 68 — Western Mediterranean, reachable from Gibraltar.
	ai_strategy = {
		type = naval_avoid_region
		id = 68
		value = 200
	}

	# 69 — Eastern Mediterranean, which vanilla ENG uses as a Royal Navy home
	# base (strike_force_home_base id = 69).
	ai_strategy = {
		type = naval_avoid_region
		id = 69
		value = 250
	}
}


# ===========================================================================
# SOVIET UNION
# ===========================================================================
#
# The Soviet navy is four separate fleets that cannot reinforce each other, all
# of them small. It has no business in open water. Vanilla already avoids the
# Sea of Japan (id = 79, value 150); this extends the same logic to the north
# and keeps the Baltic and Black Sea fleets at home where they are at least a
# coastal defence.

snai_sov_coastal_defence_only = {
	allowed = {
		original_tag = SOV
	}
	enable = {
		has_war = yes
	}
	abort_when_not_enabled = yes

	# 16 — North Sea, 45 — Norwegian Sea, 44 — Denmark Strait: nothing Soviet
	# needs to be in any of them.
	ai_strategy = {
		type = naval_avoid_region
		id = 16
		value = 400
	}

	ai_strategy = {
		type = naval_avoid_region
		id = 45
		value = 300
	}

	ai_strategy = {
		type = naval_avoid_region
		id = 44
		value = 300
	}

	# 30 — Black Sea, and 9 — Baltic Sea: the two waters it can actually defend.
	ai_strategy = {
		type = strike_force_home_base
		id = 30
		value = 100
	}

	ai_strategy = {
		type = strike_force_home_base
		id = 9
		value = 100
	}

	ai_strategy = {
		type = role_ratio
		id = naval_capital_bb
		value = -60
	}

	ai_strategy = {
		type = role_ratio
		id = naval_screen
		value = 30
	}
}
