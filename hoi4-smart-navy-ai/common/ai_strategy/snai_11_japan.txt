# Smart Navy AI — Japan
#
# WHAT VANILLA ALREADY DOES, AND WHY IT IS NOT ENOUGH
#
# Vanilla 1.14.1 JAP.txt contains exactly one naval mission strategy in the
# whole file, jap_conserve_fuel_for_usa_fight:
#
#     enable = { date < 1941.1.1  NOT = { has_war_with = USA } }
#     ai_strategy = { type = naval_mission_threshold
#                     id = "MISSION_CONVOY_ESCORT" value = 150 }
#
# That is a fuel-saving measure and it is correct on its own terms — nobody is
# raiding Japan in 1940, so escorting is wasted fuel. This mod deliberately
# leaves it alone.
#
# The problem is what happens after it aborts. On war with the United States the
# plan switches off and Japan reverts to the base escort threshold of 100 — an
# ordinary, unremarkable willingness to escort — at the exact moment the largest
# submarine campaign of the war opens against the one sea lane Japan cannot
# live without. There is no plan anywhere in vanilla JAP.txt that raises escort
# priority, builds escorts, or tells the fleet which water actually matters.
#
# Japan does not lose because its fleet is beaten. It loses because its oil
# stops arriving while the fleet is somewhere in the central Pacific.


# ---------------------------------------------------------------------------
# The oil lane is the war.
#
# The route that matters runs from the Dutch East Indies to the Home Islands:
# Java Sea and Celebes Sea, up through the South China Sea, past Formosa into
# the East China Sea, and home. Every one of those is a narrow, shallow,
# escortable water — which is the good news, because it means escorts work.
#
# The threshold is pushed hard negative here rather than merely lowered. Convoy
# escort is the mission Japan should almost always be running, and it is
# competing against the AI's standing preference for hunting the enemy fleet.
# ---------------------------------------------------------------------------
snai_jap_the_oil_lane_is_the_war = {
	allowed = {
		original_tag = JAP
	}
	enable = {
		has_war_with = USA
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_ESCORT"
		value = -90
	}

	ai_strategy = {
		type = role_ratio
		id = naval_escort
		value = 60
	}

	ai_strategy = {
		type = unit_ratio
		id = convoy
		value = 30
	}

	# 75 — South China Sea. The middle of the lane, and within land-based air
	# cover from Formosa, Indochina and the Philippines for most of its length.
	# Basing the strike force here keeps it where the shipping is instead of
	# leaving it parked in the Home Islands.
	ai_strategy = {
		type = strike_force_home_base
		id = 75
		value = 140
	}

	# 76 — East China Sea, the last leg into the Home Islands.
	ai_strategy = {
		type = strike_force_home_base
		id = 76
		value = 100
	}
}


# ---------------------------------------------------------------------------
# Do not chase the American fleet across the Pacific.
#
# The central and eastern Pacific is where the United States wants the battle:
# its carriers, its land-based air on the island chains, and its repair capacity
# are all closer than Japan's. Every region below sits outside Japanese land-based
# air cover and inside American.
#
# Expert AI reaches the same conclusion independently — it ships
# naval_avoid_region entries for Hawaii (105) and Midway (95) — which is a
# useful cross-check that these ids are right and that the judgement is not just
# this mod's opinion.
#
# Released if Japan genuinely outmatches its enemies at sea, so a player-shaped
# or lucky Japan is not held back by it.
# ---------------------------------------------------------------------------
snai_jap_do_not_chase_the_us_fleet = {
	allowed = {
		original_tag = JAP
	}
	enable = {
		has_war_with = USA
		enemies_naval_strength_ratio < 1.2
	}
	abort_when_not_enabled = yes

	# 105 — Hawaii Ridge.
	ai_strategy = {
		type = naval_avoid_region
		id = 105
		value = 400
	}

	# 95 — West Emperor Chain (Midway).
	ai_strategy = {
		type = naval_avoid_region
		id = 95
		value = 300
	}

	# 111 — North Central Pacific.
	ai_strategy = {
		type = naval_avoid_region
		id = 111
		value = 250
	}

	# 176 — Central North Pacific.
	ai_strategy = {
		type = naval_avoid_region
		id = 176
		value = 250
	}

	# 114 — North Pacific, and 88 — Bering Sea. Nothing Japan needs is up there,
	# and it is a long way from any friendly port.
	ai_strategy = {
		type = naval_avoid_region
		id = 114
		value = 200
	}

	ai_strategy = {
		type = naval_avoid_region
		id = 88
		value = 200
	}
}


# ---------------------------------------------------------------------------
# Once the carriers are gone, stop pretending.
#
# A Japan below four carriers has lost the instrument it fights with. From that
# point the fleet is worth more covering the lane than hunting, and the
# remaining hulls are worth more alive than traded.
# ---------------------------------------------------------------------------
snai_jap_after_the_carriers = {
	allowed = {
		original_tag = JAP
	}
	enable = {
		has_war_with = USA
		date > 1942.6.1
		NOT = {
			has_navy_size = {
				type = carrier
				size > 3
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_STRIKE_FORCE"
		value = 250
	}

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_ESCORT"
		value = -60
	}

	# 78 — Philippine Sea. The outer edge of what is still defensible with
	# land-based air, and the water an invasion of the Home Islands must cross.
	ai_strategy = {
		type = strike_force_home_base
		id = 78
		value = 120
	}

	ai_strategy = {
		type = role_ratio
		id = naval_escort
		value = 40
	}

	ai_strategy = {
		type = role_ratio
		id = naval_capital_bb
		value = -50
	}
}


# ---------------------------------------------------------------------------
# Take the resource area, then hold it.
#
# east_indies and home_islands are named ai_areas defined in common/ai_areas/
# and both are used by vanilla, so they are safe ids. area_priority takes these
# names — it does not take strategic region numbers.
# ---------------------------------------------------------------------------
snai_jap_southern_resource_area = {
	allowed = {
		original_tag = JAP
	}
	enable = {
		has_war_with = USA
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = area_priority
		id = east_indies
		value = 80
	}

	ai_strategy = {
		type = area_priority
		id = home_islands
		value = 60
	}
}
