# Kekistan is not a fascist state.
#
# GER starts 1936 as fascist Germany in vanilla, and its history file is huge
# (army, navy, ideas, dozens of characters), so rather than replacing that file
# and risking breaking all of it, the politics are flipped once on startup.

on_actions = {
	on_startup = {
		effect = {
			GER = {
				set_politics = {
					ruling_party = democratic
					elections_allowed = yes
					last_election = "1936.1.1"
				}

				set_popularities = {
					democratic = 70
					neutrality = 15
					communism = 10
					fascism = 5
				}

				create_country_leader = {
					name = "Ben Shapiro"
					desc = "POLITICS_KEKISTAN_LEADER_DESC"
					picture = "Portrait_Kekistan_Ben_Shapiro.dds"
					expire = "1965.1.1"
					ideology = conservatism
					traits = {}
				}
			}
		}
	}
}
